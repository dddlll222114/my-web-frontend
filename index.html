int start_angle = 0; for (int i = 0; i < current_option_count; i++) { int angle = (int)(current_counts[i] * 360.0 / total); HBRUSH brush = CreateSolidBrush(colors[i % 6]); SelectObject(hdc, brush); Pie(hdc, pie_cx - r, pie_cy - r, pie_cx + r, pie_cy + r, pie_cx + (int)(r * cos(start_angle * 3.14159 / 180)), pie_cy - (int)(r * sin(start_angle * 3.14159 / 180)), pie_cx + (int)(r * cos((start_angle + angle) * 3.14159 / 180)), pie_cy - (int)(r * sin((start_angle + angle) * 3.14159 / 180))); DeleteObject(brush); start_angle += angle; } } // 2. 文字画在右侧，不重叠 int text_x = pie_cx + r + 20; int text_y = 15; char text[256] = { 0 }; sprintf_s(text, "状态：%s", vote_status); TextOutA(hdc, text_x, text_y, text, (int)strlen(text)); text_y += 25; for (int i = 0; i < current_option_count; i++) { sprintf_s(text, "%s : %d 票", option_names[i], current_counts[i]); TextOutA(hdc, text_x, text_y, text, (int)strlen(text)); text_y += 25; } EndPaint(hdcWnd, &ps);
